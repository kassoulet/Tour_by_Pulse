palette and doubleimage — the 320x240 "low-res" path

The Medusa/Reebok pictures are drawn as 8-bit indices in a 320x240 buffer (TEMP8, filled by fillbuffer, echo5 or reebok). palette expands the indices to 32-bit colour through a 256-entry table made from a 768-byte VGA palette file (6-bit components, so every value is multiplied by four), and doubleimage blows the 320x240 result up to 640x480 by writing every pixel twice on two rows.

The table is rebuilt every frame in the original (the loop at 0x108e3d is cheap); the whole conversion (r<<16 | g<<8 | b) << 2 is done on the packed dword, which only works because the components are at most 63. For a 16-bit destination the same table is built as RGB565 from the 6-bit values (r>>1, g, b>>1). The doubling routine (0x1e896) has the pitch (0xa00 = 640*4 bytes) and the width (0x140 pixels) burnt in: it is a one-trick routine, and each iteration moves 8 bytes in and writes 4x8.

; palette run, 32-bit destination  (0x108e2d)  edi=dst esi=src ecx=w*h ebx=palette file
108e3d  mov dl, [ebx+2]            ; b
108e40  mov al, [ebx]              ; r
108e42  shl eax, 0x10
108e45  mov dh, [ebx+1]            ; g
108e48  or edx, eax                ; 0x00RRGGBB (6-bit components)
108e4a  add ebx, 3
108e4d  shl edx, 2                 ; all three components x4 at once
108e52  mov [ebp*4+0x1b7b94], edx  ; table[-256..-1]  (0x1b7794)
108e59  inc ebp
108e5f  jne 0x108e3d
108e61  mov al, [esi]              ; --- pixel loop: one table read per pixel
108e63  inc esi
108e64  add edi, 4
108e68  mov edx, [eax*4+0x1b7794]
108e6f  dec ecx
108e70  mov [edi-4], edx
108e73  jne 0x108e61

; doubleimage inner loop (0x1e89e)   esi=src, edi=dst, one pass = 2 source pixels -> 2x2 block each
1e89e  mov eax, [esi]
1e8a0  mov ebx, [esi+4]
1e8a3  mov [edi], eax              ; row 0: p0 p0
1e8a5  mov [edi+8], ebx            ;        p1 p1
1e8a8  mov [edi+4], eax
1e8ab  mov [edi+0xc], ebx
1e8ae  mov [edi+0xa00], eax        ; row 1 (+640*4)
1e8b4  mov [edi+0xa08], ebx
1e8ba  mov [edi+0xa04], eax
1e8c0  mov [edi+0xa0c], ebx
1e8c6  add esi, 8
1e8cc  add edi, 0x10
1e8d2  sub ecx, 2
1e8d8  jne 0x1e89e

t32[i] = (((r << 16) | (g << 8) | b) << 2) >>> 0;          // palette file bytes are 6-bit
